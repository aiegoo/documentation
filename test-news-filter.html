Test: Simple News Filter

Method 1: Basic loop through site.wiki

{%- for p in site.wiki -%} {%- if p.auto_imported == true -%}

{{ p.title | default: p.name }}

Tags: {{ p.tags | join: ", " }}

Date: {{ p.date | default: p.updated }}

Auto-imported: {{ p.auto_imported }}

{%- endif -%} {%- endfor -%}

Method 2: Filter by tags (exact same logic as news.html)

{%- assign wiki_sorted = site.wiki | sort: "date" | reverse -%} {%- for p in wiki_sorted -%} {%- if p.tags and (p.tags contains 'news' or p.tags contains 'updates' or p.tags contains 'announcements') -%}

{{ p.title | default: p.name }}

Tags: {{ p.tags | join: ", " }}

Date: {{ p.date | default: p.updated }}

Auto-imported: {{ p.auto_imported }}

Source file: {{ p.source_file }}

{%- endif -%} {%- endfor -%}

Method 3: Check specific tags individually

{%- for p in site.wiki -%} {%- if p.auto_imported == true -%}

{{ p.title | default: p.name }}

All tags: {{ p.tags | join: " | " }}

Individual checks:

  • news: {{ p.tags contains 'news' }}
  • updates: {{ p.tags contains 'updates' }}
  • announcements: {{ p.tags contains 'announcements' }}

Combined condition: {{ p.tags and (p.tags contains 'news' or p.tags contains 'updates' or p.tags contains 'announcements') }}

{%- endif -%} {%- endfor -%}
